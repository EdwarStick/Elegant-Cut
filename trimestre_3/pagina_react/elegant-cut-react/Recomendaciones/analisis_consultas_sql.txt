11. 🔄 PROCEDIMIENTOS ALMACENADOS
    Para operaciones complejas frecuentes:
    
    DELIMITER //
    CREATE PROCEDURE sp_agendar_cita(
        IN p_nombre VARCHAR(100),
        IN p_telefono VARCHAR(20),
        IN p_email VARCHAR(100),
        IN p_fecha DATE,
        IN p_hora_inicio INT,
        IN p_servicio_id INT,
        IN p_observaciones TEXT,
        OUT p_reserva_id INT
    )
    BEGIN
        DECLARE v_usuario_id INT;
        DECLARE v_horario_id INT;
        
        START TRANSACTION;
        
        -- Buscar o crear usuario
        SELECT id_usuario INTO v_usuario_id 
        FROM usuarios 
        WHERE telefono = p_telefono OR email = p_email
        LIMIT 1;
        
        IF v_usuario_id IS NULL THEN
            INSERT INTO usuarios (prim_nombre, telefono, email, id_rol, estado, created_at)
            VALUES (p_nombre, p_telefono, p_email, 3, 1, NOW());
            SET v_usuario_id = LAST_INSERT_ID();
        END IF;
        
        -- Obtener horario
        SELECT id_horarios INTO v_horario_id
        FROM horarios
        WHERE hora_inicio = p_hora_inicio;
        
        -- Crear reserva
        INSERT INTO reservas (fecha, observaciones, id_usuario, id_estado_cita, id_horarios)
        VALUES (p_fecha, p_observaciones, v_usuario_id, 1, v_horario_id);
        SET p_reserva_id = LAST_INSERT_ID();
        
        -- Asociar servicio
        INSERT INTO detalle_cita_servicio (id_reservas, id_servicio)
        VALUES (p_reserva_id, p_servicio_id);
        
        COMMIT;
    END //
    DELIMITER ;
